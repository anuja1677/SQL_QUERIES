Level 1: Basic SQL queries

1. Display all employee details.
ANS : select * from emp;

2. Display employee names and salaries.
ANS:  select ename, sal
      from emp;

3. Find employees whose salary is greater than 3000.
ANS: 
 SELECT ENAME
 FROM EMP
 WHERE SAL = 3000;

4. Display employees who work in department 10.
ANS:
 	SELECT ENAME
	FROM EMP
	WHERE DEPTNO = 10;

5. Find employees whose names start with 'S'.
ANS:
	SELECT ENAME 
	FROM EMP
	WHERE ENAME LIKE 'S%';

6. Display employees in descending order of salary.
ANS:
	SELECT ENAME
	FROM EMP
	ORDER BY SAL DESC;

7. Find employees whose salary is between 2000 and 5000.
ANS:
	SELECT ENAME
	FROM EMP
	WHERE SAL BETWEEN 2000 AND 5000;

8. Display employees who receive commission (COMM is not NULL).
ANS:    SELECT ENAME 
 	FROM EMP
 	WHERE COMM IS NOT NULL;

Level 2: Aggregate functions and GROUP BY

1. Find the highest salary in the company
ANS:
	SELECT MAX(SAL)
 	FROM EMP;

2. Find the second-highest salary.
ANS:
	SELECT MAX(SAL)
 	FROM EMP
 	WHERE SAL < (SELECT MAX(SAL)
              	      FROM EMP);
OR

SELECT SAL
FROM (SELECT EMP.*,DENSE_RANK()OVER (ORDER BY SAL DESC)
      RANK
      FROM EMP)
WHERE RANK IN (2);

3. Count the total number of employees.
ANS:
	SELECT COUNT(*)
	FROM EMP;
	
4. Find the average salary of employees.
ANS:
	SELECT AVG(SAL)
	FROM EMP;

5. Display department-wise employee counts.
ANS:
	SELECT COUNT(*),DEPTNO
	FROM EMP
	GROUP BY DEPTNO;

6. Find departments having more than 3 employees.
ANS:
	SELECT DEPTNO
	FROM EMP
	GROUP BY DEPTNO
	HAVING COUNT(*)>3;

7. Display job-wise maximum salary.
ANS:
	SELECT MAX(SAL) , JOB
	FROM EMP
	GROUP BY JOB;

8. Find jobs in which more than 2 employees work.
ANS:
	SELECT JOB
 	FROM EMP
 	GROUP BY JOB
 	HAVING COUNT(*)>2;

Level 3: JOIN queries

1. Display employee names with their department names.
ANS:
	SELECT ENAME,DNAME
	FROM EMP INNER JOIN DEPT
	ON EMP.DEPTNO = DEPT.DEPTNO;

2. Display employees working in the Research department.
ANS:
	SELECT ENAME
 	FROM EMP INNER JOIN DEPT
 	ON EMP.DEPTNO = DEPT.DEPTNO
 	WHERE DNAME = 'RESEARCH';

3. Display departments that have no employees.
ANS:
	
	SELECT DNAME
 	FROM EMP RIGHT JOIN DEPT
 	ON EMP.DEPTNO = DEPT.DEPTNO
 	WHERE EMP.DEPTNO IS NULL;

4. Display employee names with their manager names (self join).
ANS:
	SELECT E1.ENAME,E2.ENAME
 	FROM EMP E1 JOIN EMP E2
 	ON E1.MGR = E2.EMPNO;

5. Count employees in each department and display department names.
ANS:
	SELECT COUNT(*),DNAME
	FROM EMP INNER JOIN DEPT
	ON EMP.DEPTNO = DEPT.DEPTNO
	GROUP BY DNAME;

6. Display employees who do not belong to any matching department.
ANS:
	SELECT ENAME
	FROM EMP LEFT JOIN DEPT
	ON EMP.DEPTNO = DEPT.DEPTNO
	WHERE DEPT.DEPTNO IS NULL;

Level 4: Subqueries — interview favorites

1. Find employees earning more than the average salary.
ANS:
	SELECT ENAME
	FROM EMP
	WHERE SAL > (SELECT AVG(SAL)
                     FROM EMP);

2. Find employees earning the highest salary in each department.
ANS:
	SELECT ENAME
	FROM EMP
	WHERE SAL IN (SELECT MAX(SAL)
           		FROM EMP
           		GROUP BY DEPTNO);

3. Find employees earning more than ALL employees in department 10.
ANS:
	SELECT ENAME
 	FROM EMP
 	WHERE SAL > ALL(SELECT SAL
                 FROM EMP
                 WHERE DEPTNO = 10);

4. Find employees earning more than ANY employee in department 10.
ANS:
	SELECT ENAME
	FROM EMP
	WHERE SAL >ANY(SELECT SAL
               		FROM EMP
               		WHERE DEPTNO = 10);

5. Find the second-highest salary using a subquery.
ANS:
	SELECT MAX(SAL)
	FROM EMP
	WHERE SAL < (SELECT MAX(SAL)
             		FROM EMP);
OR
	SELECT SAL
	FROM (SELECT EMP.*,DENSE_RANK()OVER(ORDER BY SAL DESC)RANK
		FROM EMP)
	WHERE RANK IN 2;

6. Find employees who earn more than their own department's average salary.
ANS:
	SELECT ENAME
	FROM EMP E
	WHERE SAL > (SELECT AVG(SAL)
             		FROM EMP
             		WHERE DEPTNO = E.DEPTNO);

7. Find the employee(s) who earn the third-highest distinct salary.
ANS:
	SELECT ENAME
	FROM EMP
	WHERE SAL IN (SELECT MAX(DISTINCT SAL)
              		FROM EMP
              		WHERE SAL < (SELECT MAX(DISTINCT SAL)
                           		FROM EMP
                           		WHERE SAL <(SELECT MAX(DISTINCT SAL)
                                       			FROM EMP)));

8. Find employees who earn more than their manager.
ANS:
	SELECT E1.ENAME
	FROM EMP E1 JOIN EMP E2
	ON E1.MGR = E2.EMPNO
	WHERE E1.SAL > E2.SAL;

9. Find the department with the highest average salary.
ANS:
	SELECT DNAME,AVG(SAL)
	FROM EMP
	INNER JOIN DEPT
	ON EMP.DEPTNO = DEPT.DEPTNO
	GROUP BY DNAME
	HAVING AVG(SAL)IN (SELECT MAX(AVG_SAL)
                 FROM (SELECT AVG(SAL) AVG_SAL
                   FROM EMP
                   GROUP BY DEPTNO));

10. display employees who earn the same salary as another employee.
ANS:
	 SELECT DISTINCT E1.ENAME
 	FROM EMP E1 JOIN EMP E2
 	ON E1.SAL = E2.SAL
 	WHERE E1.EMPNO != E2.EMPNO;

11. Find employees who earn the second-highest salary in each department.
ANS:
	 SELECT ENAME
 	FROM EMP E1
 	WHERE SAL IN (SELECT SAL
              		FROM (SELECT EMP.*,DENSE_RANK()OVER (ORDER BY SAL DESC)RANK
                       		FROM EMP)
              		WHERE RANK IN 2
              		AND DEPTNO = E1.DEPTNO);

12. WAQTD EMP name and department name and location for employees whose name start with 'A'.
ANS:
	SELECT ENAME,DNAME,LOC
 	FROM EMP INNER JOIN DEPT
 	ON EMP.DEPTNO = DEPT.DEPTNO
 	WHERE SUBSTR(ENAME,1,1) = 'A'

13.WAQTD EMP name and department name ,manager name and his location where the second character of the emp name and 2nd character of manager name should be same(without using like operator).
ANS:
	SELECT E1.ENAME,D1.DNAME,E2.ENAME,D2.LOC
	FROM EMP E1 JOIN EMP E2
	ON E1.MGR = E2.EMPNO JOIN DEPT D1 ON E1.DEPTNO = D1.DEPTNO JOIN DEPT D2 ON E2.DEPTNO = D2.DEPTNO
	WHERE SUBSTR(E1.ENAME,2,1) = SUBSTR(E2.ENAME,2,1);

14.WAQTD EMP name, sal and deptno where the emp is earning more than the avg sal of their department and loc contains only 5 character.
ANS:
	SELECT ENAME ,SAL,EMP.DEPTNO
 	FROM EMP INNER JOIN DEPT
 	ON EMP.DEPTNO = DEPT.DEPTNO
 	WHERE LENGTH(LOC) = 5 AND SAL IN (SELECT AVG(E.SAL)
                            			FROM EMP E
                            			WHERE E.DEPTNO = EMP.DEPTNO);

15. WAQTD dname and location  where the maximum employees  are their the particular dept
ANS:
	SELECT DEPT.DNAME,DEPT.LOC
	FROM DEPT INNER JOIN EMP
	ON DEPT.DEPTNO = EMP.DEPTNO
	GROUP BY DEPT.DEPTNO ,DEPT.DNAME,DEPT.LOC
	HAVING COUNT(*) IN (SELECT MAX(COUNT)
                    		FROM (SELECT COUNT(*) COUNT
                           	FROM EMP
                           	GROUP BY DEPTNO));
